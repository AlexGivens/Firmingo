cmake_minimum_required(VERSION 3.20)
project(Firmingo LANGUAGES C CXX)
include(CTest)
option(FIRMINGO_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(FIRMINGO_DHCP_FUZZ "Build the separate bounded DHCP mutation job" OFF)
option(FIRMINGO_PROTOCOL_FUZZ "Build the separate bounded protocol parser mutation job" OFF)

add_library(firmingo_core src/core/stream.cpp src/core/protocol.cpp src/core/json.cpp src/core/session.cpp src/core/uart.cpp)
target_include_directories(firmingo_core PUBLIC src)
target_compile_features(firmingo_core PUBLIC cxx_std_11)
set_target_properties(firmingo_core PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(firmingo_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(FIRMINGO_SANITIZE)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "Sanitizers require Clang or GCC")
  endif()
  target_compile_options(firmingo_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(firmingo_core PUBLIC -fsanitize=address,undefined)
endif()
if(BUILD_TESTING)
  add_library(unity STATIC third_party/unity/src/unity.c)
  target_include_directories(unity PUBLIC third_party/unity/src)
  add_executable(stream_tests tests/native/test_stream.cpp)
  target_link_libraries(stream_tests PRIVATE firmingo_core unity)
  add_test(NAME stream COMMAND stream_tests)
  set_tests_properties(stream PROPERTIES TIMEOUT 10)
  add_executable(application_tests tests/native/test_application.cpp)
  target_link_libraries(application_tests PRIVATE firmingo_core unity)
  add_test(NAME application COMMAND application_tests)
  set_tests_properties(application PROPERTIES TIMEOUT 10)
  add_executable(uart_tests tests/native/test_uart.cpp)
  target_link_libraries(uart_tests PRIVATE firmingo_core unity)
  add_test(NAME uart COMMAND uart_tests)
  set_tests_properties(uart PROPERTIES TIMEOUT 10)
  add_executable(protocol_tests tests/native/test_protocol.cpp)
  target_compile_definitions(protocol_tests PRIVATE FIXTURE_DIR="${CMAKE_SOURCE_DIR}/tests/fixtures/protocol-v1")
  target_link_libraries(protocol_tests PRIVATE firmingo_core unity)
  add_test(NAME protocol COMMAND protocol_tests)
  set_tests_properties(protocol PROPERTIES TIMEOUT 10)
  add_executable(session_tests tests/native/test_session.cpp)
  target_compile_definitions(session_tests PRIVATE FIXTURE_DIR="${CMAKE_SOURCE_DIR}/tests/fixtures/protocol-v1")
  target_link_libraries(session_tests PRIVATE firmingo_core unity)
  add_test(NAME session COMMAND session_tests)
  set_tests_properties(session PROPERTIES TIMEOUT 10)
  if(FIRMINGO_PROTOCOL_FUZZ)
    add_executable(protocol_fuzz tests/native/fuzz_protocol.cpp)
    target_link_libraries(protocol_fuzz PRIVATE firmingo_core)
    add_test(NAME protocol_fuzz COMMAND protocol_fuzz)
    set_tests_properties(protocol_fuzz PROPERTIES TIMEOUT 20)
    add_executable(session_fuzz tests/native/fuzz_session.cpp)
    target_link_libraries(session_fuzz PRIVATE firmingo_core)
    add_test(NAME session_fuzz COMMAND session_fuzz)
    set_tests_properties(session_fuzz PROPERTIES TIMEOUT 20)
  endif()
  add_executable(tcp_echo_tests tests/native/test_tcp_echo.cpp src/ports/arduino_pico/tcp_echo.cpp)
  target_include_directories(tcp_echo_tests PRIVATE tests/native/fake_tcp)
  target_link_libraries(tcp_echo_tests PRIVATE firmingo_core unity)
  add_test(NAME tcp_echo COMMAND tcp_echo_tests)
  set_tests_properties(tcp_echo PROPERTIES TIMEOUT 10)
  add_executable(tcp_session_tests tests/native/test_tcp_session.cpp src/ports/arduino_pico/tcp_session.cpp)
  target_include_directories(tcp_session_tests PRIVATE tests/native/fake_tcp)
  target_compile_options(tcp_session_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(tcp_session_tests PRIVATE firmingo_core unity)
  add_test(NAME tcp_session COMMAND tcp_session_tests)
  set_tests_properties(tcp_session PROPERTIES TIMEOUT 10)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(usb_patch_dir "${CMAKE_SOURCE_DIR}/patches/arduino-pico-6.0.0-usb-startup")
  file(GLOB_RECURSE usb_startup_sources CONFIGURE_DEPENDS "third_party/arduino-pico-usb-startup/upstream/*")
  add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/usb_startup.inc"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/native/generate_usb_startup.py"
      "${CMAKE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/usb_startup.inc"
    DEPENDS tests/native/generate_usb_startup.py examples/firmingo_local_only/firmingo_local_only.ino examples/firmingo_application/firmingo_application.ino
      "${usb_patch_dir}/deferred-ncm.patch" "${usb_patch_dir}/manifest.json" ${usb_startup_sources})
  add_custom_target(usb_startup_fragments DEPENDS "${CMAKE_BINARY_DIR}/usb_startup.inc")
  foreach(profile deferred default)
    add_executable(usb_startup_${profile}_tests tests/native/test_usb_startup.cpp)
    add_dependencies(usb_startup_${profile}_tests usb_startup_fragments)
    target_include_directories(usb_startup_${profile}_tests PRIVATE "${CMAKE_BINARY_DIR}")
    target_link_libraries(usb_startup_${profile}_tests PRIVATE firmingo_core unity)
    if(profile STREQUAL "deferred")
      target_compile_definitions(usb_startup_${profile}_tests PRIVATE FIRMINGO_USB_DEFERRED_START DISABLE_USB_SERIAL)
    endif()
    add_test(NAME usb_startup_${profile} COMMAND usb_startup_${profile}_tests)
    set_tests_properties(usb_startup_${profile} PROPERTIES TIMEOUT 10)
  endforeach()
  add_library(local_dhcp STATIC src/ports/arduino_pico/local_dhcp.c)
  target_include_directories(local_dhcp PUBLIC src tests/native/fake_lwip)
  target_compile_options(local_dhcp PRIVATE -Wall -Wextra -Werror)
  if(FIRMINGO_SANITIZE)
    target_compile_options(local_dhcp PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(local_dhcp PUBLIC -fsanitize=address,undefined)
  endif()
  add_executable(dhcp_tests tests/native/test_dhcp.c)
  target_link_libraries(dhcp_tests PRIVATE local_dhcp unity)
  add_test(NAME local_dhcp COMMAND dhcp_tests)
  set_tests_properties(local_dhcp PROPERTIES TIMEOUT 10)
  if(FIRMINGO_DHCP_FUZZ)
    add_executable(dhcp_fuzz tests/native/test_dhcp.c)
    target_compile_definitions(dhcp_fuzz PRIVATE FIRMINGO_DHCP_FUZZ=1)
    target_link_libraries(dhcp_fuzz PRIVATE local_dhcp unity)
    add_test(NAME dhcp_fuzz COMMAND dhcp_fuzz)
    set_tests_properties(dhcp_fuzz PROPERTIES TIMEOUT 20)
  endif()
endif()
